cmake_minimum_required(VERSION 3.24)
project(LanBridge LANGUAGES CXX)
enable_language(RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bridge STATIC src/Bridge.cpp src/Clipboard.cpp src/Pairing.cpp)
target_include_directories(bridge PUBLIC src)
target_compile_definitions(bridge PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(bridge PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(bridge PUBLIC ws2_32 bcrypt crypt32 iphlpapi user32 shell32 ole32)
add_library(shelltransfer STATIC src/ShellTransfer.cpp)
target_include_directories(shelltransfer PUBLIC src)
target_compile_definitions(shelltransfer PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(shelltransfer PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(shelltransfer PUBLIC user32 shell32 ole32 uuid)
add_library(LanBridgeShell SHARED src/ShellCommand.cpp src/ShellCommand.def src/ShellCommand.rc)
target_compile_options(LanBridgeShell PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(LanBridgeShell PRIVATE shelltransfer shlwapi)
add_executable(LanBridge WIN32 src/Main.cpp src/App.rc third_party/qrcodegen/qrcodegen.cpp)
add_library(startup STATIC src/Startup.cpp)
target_compile_options(startup PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(startup PUBLIC shelltransfer advapi32)
add_library(ui STATIC src/ui/TitleBar.cpp src/ui/PopupMenu.cpp)
target_include_directories(ui PUBLIC src)
target_compile_definitions(ui PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ui PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(ui PUBLIC user32 gdi32 comctl32 shell32)
target_include_directories(LanBridge PRIVATE third_party/qrcodegen)
target_compile_options(LanBridge PRIVATE /W4 /EHsc /utf-8 /permissive-)
target_link_libraries(LanBridge PRIVATE bridge shelltransfer startup ui comctl32 comdlg32 gdi32 dwmapi)
target_link_options(LanBridge PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
# Relink when the embedded manifest changes (including package identity removal).
set_property(TARGET LanBridge APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
include(CTest)
if(BUILD_TESTING)
  add_executable(PairingIntegration tests/PairingIntegration.cpp)
  target_link_libraries(PairingIntegration PRIVATE bridge shelltransfer)
  target_compile_options(PairingIntegration PRIVATE /utf-8 /EHsc)

  set_tests_properties(PairingIntegration PROPERTIES TIMEOUT 30)
  add_test(NAME RememberedDesktopIntegration COMMAND PairingIntegration --desktop $<TARGET_FILE:LanBridge>)
  set_tests_properties(RememberedDesktopIntegration PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
  add_executable(TypographyPreview tests/TypographyPreview.cpp)
  target_link_libraries(TypographyPreview PRIVATE ui gdiplus)
  target_compile_options(TypographyPreview PRIVATE /utf-8 /EHsc)
  add_executable(StartupIntegration tests/StartupIntegration.cpp)
  target_link_libraries(StartupIntegration PRIVATE bridge startup)
  target_compile_options(StartupIntegration PRIVATE /utf-8 /EHsc)
  add_test(NAME StartupIntegration COMMAND StartupIntegration $<TARGET_FILE:LanBridge>)
  set_tests_properties(StartupIntegration PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
  add_executable(ShellIntegration tests/ShellIntegration.cpp)
  target_link_libraries(ShellIntegration PRIVATE bridge shelltransfer)
  target_compile_options(ShellIntegration PRIVATE /utf-8 /EHsc)
  add_test(NAME ShellIntegration COMMAND ShellIntegration $<TARGET_FILE:LanBridge> $<TARGET_FILE:LanBridgeShell>)
